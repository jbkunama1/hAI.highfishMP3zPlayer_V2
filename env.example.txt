# MP3z Player — Konfiguration
# Lokal:    cp env.example.txt .env && joe .env
# Portainer: dieselben Namen als Stack-Environment-Variablen eintragen

# ── Server ───────────────────────────────────────────────────────
HOST=0.0.0.0
PORT=80

# ── Zugriff (Login-Passwort der Web-App = API-Key) ──────────────────────
# Empfohlen: eigenen, langen Schlüssel setzen. Ist er leer, gilt SMB_PASS als Key
# (im SMB-Modus NICHT empfohlen: der Key taucht in Stream-URLs/Logs auf).
API_KEY_OVERRIDE=
DEFAULT_URL=http://localhost:8066

# ── Musikquelle: local ODER smb ───────────────────────────────────
STORAGE_MODE=local

# --- STORAGE_MODE=local ---
# Pfad im Container (Docker: Host-Pfad wird per MUSIC_HOST_PATH nach /music gemountet)
MUSIC_ROOT=/mnt/USBHDD_MP3z
# Nur Docker/Portainer: Host-Verzeichnis
MUSIC_HOST_PATH=/mnt/USBHDD_MP3z

# --- STORAGE_MODE=smb ---
SMB_HOST=
SMB_SHARE=
# optional: Unterordner innerhalb der Freigabe, z.B. Musik/MP3z
SMB_PATH=
SMB_USER=
SMB_PASS=
# optional: Windows-Domäne/Workgroup
SMB_DOMAIN=
SMB_PORT=445
# true = SMB3-Verschlüsselung erzwingen (Server muss das unterstützen)
SMB_ENCRYPT=false

# ── Docker: Benutzer, unter dem der Container läuft (wichtig für rw-Mount) ─────
PUID=1000
PGID=1000
HOST_PORT=8066
