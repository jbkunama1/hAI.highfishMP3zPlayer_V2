# MP3z Samba Player — Konfiguration
# Kopiere diese Datei nach .env und trage deine Werte ein:
#   cp .env.example .env
#   joe .env

# ── Server ────────────────────────────────────────────
HOST=0.0.0.0
PORT=80

# ── Musikverzeichnis ──────────────────────────────────
MUSIC_ROOT=/mnt/USBHDD_MP3z

# ── Samba-Zugangsdaten ────────────────────────────────
# (Gleiche Credentials wie deine Samba-Freigabe)
SMB_USER=dein_benutzername
SMB_PASS=dein_samba_passwort

# ── Optionaler separater API-Key ──────────────────────
# Leer lassen → SMB_PASS wird verwendet
API_KEY_OVERRIDE=

# ── App-URL (für Login-Screen Voreinstellung) ─────────
DEFAULT_URL=http://samba.arbeitermili.eu

# ── MCP-Server (KI-Integration) ─────────────────────
# Der MCP-Server macht die Musikbibliothek für KI-Clients
# (Claude Desktop, Copilot, …) über das Model Context Protocol
# durchsuchbar. Der Prozess läuft als Subprozess des MP3z-Servers.
MCP_ENABLED=1       # Master-Schalter (0=aus)
MCP_AUTOSTART=1     # Beim Serverstart automatisch mitstarten
MCP_HOST=0.0.0.0    # Host für streamable HTTP MCP-Endpoint
MCP_PORT=8080       # Port für streamable HTTP MCP-Endpoint
MCP_LOG=mcp_server.log
